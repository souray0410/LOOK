class MHD_Trainer_AMP(MHD_Trainer):
    """支持自动混合精度的训练器"""
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.scaler = torch.cuda.amp.GradScaler()

    def train_step(self, input_dict: dict) -> dict:
        self.mhd_graph.train()
        self.optimizer.zero_grad(set_to_none=True)

        batch_size = self._check_batch_consistency(input_dict)
        self.mhd_graph.update_batch_size(batch_size)

        for node in self.mhd_graph.nodes:
            node.reset()
        for node_name, tensor in input_dict.items():
            node = self.mhd_graph.get_node_by_name(node_name)
            if node is None:
                continue
            node.current_state = tensor.to(self.device, non_blocking=True).detach().requires_grad_(True)

        # 前向传播在 autocast 上下文中
        with torch.amp.autocast():
            self.mhd_graph.forward()
            loss_node = self.mhd_graph.get_node_by_name(self.backward_node_name)
            loss_tensor = loss_node.current_state.mean()

        # 缩放损失并反向传播
        self.scaler.scale(loss_tensor).backward()

        # 梯度裁剪前需 unscale
        if self.grad_clip_norm and self.grad_clip_norm > 0:
            self.scaler.unscale_(self.optimizer)
            torch.nn.utils.clip_grad_norm_(self.mhd_graph.parameters(), self.grad_clip_norm)

        self.scaler.step(self.optimizer)
        self.scaler.update()

        # 记录指标（与原始 train_step 一致）
        step_metrics = {}
        for node_name in self.monitor.monitor_nodes:
            node = self.mhd_graph.get_node_by_name(node_name)
            if node:
                val = node.current_state.detach().mean().item()
                step_metrics[node_name] = float(0.0 if np.isnan(val) or np.isinf(val) else val)

        if torch.cuda.is_available():
            torch.cuda.empty_cache()
        return step_metrics

# Cell 2: 构建多层级 3D U‑Net 图（标准 128³ 输入，InstanceNorm，bias=False 除输出层）
def build_3dunet_graph(batch_size=1, modalities=4, num_classes=4, base_ch=32, device=None):
    if device is None:
        device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    dtype = torch.float32

    def conv3d(in_ch, out_ch, ks=3, stride=1, padding=1, bias=False):
        return nn.Conv3d(in_ch, out_ch, ks, stride, padding, bias=bias)
    def in3d(out_ch):
        return nn.InstanceNorm3d(out_ch, affine=True)
    def relu():
        return nn.ReLU(inplace=True)

    class UpConcat(nn.Module):
        """上采样 + 拼接 + 双卷积（标准 UNet 解码器块）"""
        def __init__(self, in_ch1, in_ch2, out_ch):
            super().__init__()
            self.up = nn.Upsample(scale_factor=2, mode='trilinear', align_corners=False)
            self.conv = nn.Sequential(
                conv3d(in_ch1 + in_ch2, out_ch, 3, 1, 1),
                in3d(out_ch),
                relu(),
                conv3d(out_ch, out_ch, 3, 1, 1),
                in3d(out_ch),
                relu()
            )
        def forward(self, tensor_list):
            x1 = tensor_list[0]  # decoder 输入
            x2 = tensor_list[1]  # encoder 跳跃连接
            x1_up = self.up(x1)
            if x1_up.shape[2:] != x2.shape[2:]:
                x1_up = F.interpolate(x1_up, size=x2.shape[2:], mode='trilinear', align_corners=False)
            x = torch.cat([x1_up, x2], dim=1)
            return self.conv(x)

    # 节点定义（空间尺寸 128×128×128，不变）
    nodes = []
    nodes.append(MHD_Node(0, "input_img", torch.zeros(batch_size, modalities, 128, 128, 128, device=device, dtype=dtype),
                         torch.zeros(batch_size, modalities, 128, 128, 128, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(1, "label_gt", torch.zeros(batch_size, 1, 128, 128, 128, device=device, dtype=dtype),
                         torch.zeros(batch_size, 1, 128, 128, 128, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(2, "enc1", torch.zeros(batch_size, base_ch, 128, 128, 128, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch, 128, 128, 128, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(3, "enc2", torch.zeros(batch_size, base_ch*2, 64, 64, 64, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*2, 64, 64, 64, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(4, "enc3", torch.zeros(batch_size, base_ch*4, 32, 32, 32, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*4, 32, 32, 32, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(5, "enc4", torch.zeros(batch_size, base_ch*8, 16, 16, 16, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*8, 16, 16, 16, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(6, "bottleneck", torch.zeros(batch_size, base_ch*16, 8, 8, 8, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*16, 8, 8, 8, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(7, "dec1", torch.zeros(batch_size, base_ch*8, 16, 16, 16, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*8, 16, 16, 16, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(8, "dec2", torch.zeros(batch_size, base_ch*4, 32, 32, 32, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*4, 32, 32, 32, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(9, "dec3", torch.zeros(batch_size, base_ch*2, 64, 64, 64, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch*2, 64, 64, 64, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(10, "dec4", torch.zeros(batch_size, base_ch, 128, 128, 128, device=device, dtype=dtype),
                         torch.zeros(batch_size, base_ch, 128, 128, 128, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(11, "logits", torch.zeros(batch_size, num_classes, 128, 128, 128, device=device, dtype=dtype),
                         torch.zeros(batch_size, num_classes, 128, 128, 128, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(12, "loss", torch.zeros(1, 1, device=device, dtype=dtype),
                         torch.zeros(1, 1, device=device, dtype=dtype), "replace"))
    nodes.append(MHD_Node(13, "dice", torch.zeros(1, 1, device=device, dtype=dtype),
                         torch.zeros(1, 1, device=device, dtype=dtype), "replace"))

    # 边定义
    edges = []
    edge_infos = []
    def add_edge(name, ops, in_ids, out_ids, level, in_sorts=None, out_sorts=None):
        if in_sorts is None: in_sorts = [1]*len(in_ids)
        if out_sorts is None: out_sorts = [1]*len(out_ids)
        eid = len(edges)
        edges.append(MHD_Edge(eid, name, ops))
        edge_infos.append((name, in_ids, out_ids, in_sorts, level))

    # 编码器（bias=False）
    add_edge("e_enc1", [".clone()", conv3d(4, base_ch, 3,1,1), in3d(base_ch), relu(), conv3d(base_ch, base_ch, 3,1,1), in3d(base_ch), relu()], [0], [2], level=1)
    add_edge("e_enc2", [".clone()", conv3d(base_ch, base_ch*2, 3,2,1), in3d(base_ch*2), relu(), conv3d(base_ch*2, base_ch*2, 3,1,1), in3d(base_ch*2), relu()], [2], [3], level=2)
    add_edge("e_enc3", [".clone()", conv3d(base_ch*2, base_ch*4, 3,2,1), in3d(base_ch*4), relu(), conv3d(base_ch*4, base_ch*4, 3,1,1), in3d(base_ch*4), relu()], [3], [4], level=3)
    add_edge("e_enc4", [".clone()", conv3d(base_ch*4, base_ch*8, 3,2,1), in3d(base_ch*8), relu(), conv3d(base_ch*8, base_ch*8, 3,1,1), in3d(base_ch*8), relu()], [4], [5], level=4)
    add_edge("e_bottleneck", [".clone()", conv3d(base_ch*8, base_ch*16, 3,2,1), in3d(base_ch*16), relu(), conv3d(base_ch*16, base_ch*16, 3,1,1), in3d(base_ch*16), relu()], [5], [6], level=5)

    # 解码器（超边，内部已用 bias=False）
    add_edge("e_dec1", [UpConcat(base_ch*16, base_ch*8, base_ch*8)], [6,5], [7], level=6, in_sorts=[1,2])
    add_edge("e_dec2", [UpConcat(base_ch*8, base_ch*4, base_ch*4)],  [7,4], [8], level=7, in_sorts=[1,2])
    add_edge("e_dec3", [UpConcat(base_ch*4, base_ch*2, base_ch*2)],  [8,3], [9], level=8, in_sorts=[1,2])
    add_edge("e_dec4", [UpConcat(base_ch*2, base_ch, base_ch)],        [9,2], [10], level=9, in_sorts=[1,2])

    # 输出层（bias=True）
    add_edge("e_out", [".clone()", conv3d(base_ch, num_classes, 1,1,0, bias=True)], [10], [11], level=10)

    # 损失与评估
    def combined_loss_fn(tensor_list):
        logits, target = tensor_list
        ce = F.cross_entropy(logits, target.squeeze(1).long())
        probs = F.softmax(logits, dim=1)
        target_onehot = F.one_hot(target.squeeze(1).long(), num_classes=num_classes).permute(0,4,1,2,3).float()
        smooth = 1e-5
        dice = 0.0
        for c in range(1, num_classes):
            pred_c = probs[:, c, ...]
            target_c = target_onehot[:, c, ...]
            inter = (pred_c * target_c).sum()
            union = pred_c.sum() + target_c.sum()
            dice += 1.0 - (2.*inter + smooth) / (union + smooth)
        dice /= (num_classes - 1)
        return (ce + dice).reshape(1,1)

    def dice_coeff(tensor_list):
        logits, target = tensor_list
        pred = torch.softmax(logits, dim=1).argmax(1)
        target = target.squeeze(1).long()
        dices = []
        for c in range(1, num_classes):
            pred_c = (pred == c)
            target_c = (target == c)
            inter = (pred_c & target_c).sum().float()
            smooth = 1e-5
            dices.append((2.*inter + smooth) / (pred_c.sum() + target_c.sum() + smooth))
        return torch.stack(dices).mean().reshape(1,1)

    add_edge("e_loss", [combined_loss_fn], [11,1], [12], level=11, in_sorts=[1,2])
    add_edge("e_dice", [dice_coeff], [11,1], [13], level=11, in_sorts=[1,2])

    # 拓扑构建
    num_levels = max(lvl for _,_,_,_,lvl in edge_infos) + 1
    num_edges = len(edges)
    num_nodes = len(nodes)
    level_edges = [set() for _ in range(num_levels)]
    for eid, (_, _, _, _, lvl) in enumerate(edge_infos):
        level_edges[lvl].add(eid)
    def make_matrix(edge_set):
        role = torch.zeros(num_edges, num_nodes, dtype=torch.int64, device=device)
        sort = torch.zeros(num_edges, num_nodes, dtype=torch.int64, device=device)
        for eid in edge_set:
            _, in_ids, out_ids, in_sorts, _ = edge_infos[eid]
            for nid, sv in zip(in_ids, in_sorts):
                role[eid, nid] = -1
                sort[eid, nid] = sv
            for nid in out_ids:
                role[eid, nid] = 1
                sort[eid, nid] = 1
        return role, sort
    roles, sorts = [], []
    for s in level_edges:
        r, so = make_matrix(s)
        roles.append(r)
        sorts.append(so)
    topo = MHD_Topo(role_matrices=roles, sort_matrices=sorts)
    graph = MHD_Graph(nodes=set(nodes), edges=set(edges), topos={topo}, device=device)
    graph._register_all_params()
    return graph

# ---------- 标准预处理函数 ----------
def _resample_to_isotropic(data, spacing, order):
    zoom_factors = np.array(spacing) / 1.0
    return zoom(data, zoom_factors, order=order)

def _center_crop_or_pad(data, target_shape):
    current_shape = np.array(data.shape)
    target_shape = np.array(target_shape)
    if data.dtype == np.int64:
        result = np.zeros(target_shape, dtype=data.dtype)
    else:
        result = np.zeros(target_shape, dtype=np.float32)

    slices_in = []
    slices_out = []
    for i in range(3):
        if current_shape[i] >= target_shape[i]:
            start = (current_shape[i] - target_shape[i]) // 2
            end = start + target_shape[i]
            slices_in.append(slice(start, end))
            slices_out.append(slice(None))
        else:
            slices_in.append(slice(None))
            start_out = (target_shape[i] - current_shape[i]) // 2
            end_out = start_out + current_shape[i]
            slices_out.append(slice(start_out, end_out))
    result[tuple(slices_out)] = data[tuple(slices_in)]
    return result

def load_brats_case_final(path, modalities=('flair', 't1', 't1ce', 't2'),
                          target_shape=(128, 128, 128)):
    path = Path(path)
    images = []
    seg_path = None
    for f in path.glob("*.nii"):
        if 'seg' in f.name:
            seg_path = f
        else:
            images.append(f)

    sample_img = nib.load(images[0])
    orig_spacing = sample_img.header.get_zooms()[:3]

    images_sorted = []
    for mod in modalities:
        found = False
        for img_path in images:
            if mod in img_path.name:
                img_data = nib.load(img_path).get_fdata().astype(np.float32)
                img_iso = _resample_to_isotropic(img_data, orig_spacing, order=1)
                img_cropped = _center_crop_or_pad(img_iso, target_shape)
                images_sorted.append(img_cropped)
                found = True
                break
        if not found:
            raise FileNotFoundError(f"Modality {mod} not found in {path}")

    if seg_path is None:
        raise FileNotFoundError(f"Segmentation file not found in {path}")
    seg_data = nib.load(seg_path).get_fdata().astype(np.int64)
    seg_iso = _resample_to_isotropic(seg_data, orig_spacing, order=0)
    seg_cropped = _center_crop_or_pad(seg_iso, target_shape)
    seg_cropped[seg_cropped == 4] = 3

    image = np.stack(images_sorted, axis=0).astype(np.float32)
    label = seg_cropped[np.newaxis, ...]
    return torch.from_numpy(image), torch.from_numpy(label)

def compute_global_stats(case_list, base_path, modalities):
    channel_sums = np.zeros(4)
    channel_sq_sums = np.zeros(4)
    channel_counts = np.zeros(4)
    for case in tqdm(case_list, desc="Computing global stats"):
        img, _ = load_brats_case_final(os.path.join(base_path, case), modalities)
        for c in range(4):
            channel = img[c].numpy()
            foreground = channel > 0
            vals = channel[foreground]
            if len(vals) > 0:
                channel_sums[c] += vals.sum()
                channel_sq_sums[c] += (vals**2).sum()
                channel_counts[c] += len(vals)
    means = channel_sums / channel_counts
    stds = np.sqrt(channel_sq_sums / channel_counts - means**2)
    return torch.from_numpy(means).float(), torch.from_numpy(stds).float()

def global_normalize(img):
    if img.dim() == 4:
        img = img.unsqueeze(0)
        means = global_means.view(1, -1, 1, 1, 1).to(img.device)
        stds = global_stds.view(1, -1, 1, 1, 1).to(img.device)
        out = (img - means) / stds
        return out.squeeze(0)
    else:
        means = global_means.view(1, -1, 1, 1, 1).to(img.device)
        stds = global_stds.view(1, -1, 1, 1, 1).to(img.device)
        return (img - means) / stds

class MHD_Trainer_EarlyStop(MHD_Trainer_AMP):
    def __init__(self, *args, patience=50, **kwargs):
        super().__init__(*args, **kwargs)
        self.patience = patience
        self.early_stop_counter = 0
        self.stopped_epoch = 0
        
    def train(self, train_data, eval_data, epochs=500):
        self.logger.info("="*80)
        self.logger.info(f"🚀 开始训练，共 {epochs} 轮 (early stop patience={self.patience})")
        self.logger.info("="*80)
        for epoch in range(epochs):
            self.logger.info(f"\n--- Epoch {epoch+1}/{epochs} ---")
            self.train_epoch(train_data, epoch)
            prev_best = self.history["best_eval_value"]
            self.eval_epoch(eval_data, epoch)
            if self.history["best_eval_value"] > prev_best:
                self.early_stop_counter = 0
            else:
                self.early_stop_counter += 1
                if self.early_stop_counter >= self.patience:
                    self.stopped_epoch = epoch + 1
                    self.logger.info(f"Early stopping at epoch {self.stopped_epoch}")
                    break
                
            cur_ep = epoch + 1
            if self.save_interval > 0 and cur_ep % self.save_interval == 0:
                self.save_checkpoint(cur_ep)
        # 保存最终检查点（使用实际停止的 epoch）
        final_epoch = self.stopped_epoch if self.stopped_epoch > 0 else epochs
        self.save_checkpoint(final_epoch)
        self.save_training_history()
        self.logger.info("\n🎉 训练完成！")

def normalize_feat(feat, mean, std):
    if feat.dim() == 2:
        return (feat - mean.to(feat.device)) / std.to(feat.device)
    else:
        expand_dims = [1] * (feat.dim() - 2)
        mean = mean.view(1, -1, *expand_dims).to(feat.device)
        std  = std.view(1, -1, *expand_dims).to(feat.device)
        return (feat - mean) / std

def denormalize_feat(feat, mean, std):
    if feat.dim() == 2:
        return feat * std.to(feat.device) + mean.to(feat.device)
    else:
        expand_dims = [1] * (feat.dim() - 2)
        mean = mean.view(1, -1, *expand_dims).to(feat.device)
        std  = std.view(1, -1, *expand_dims).to(feat.device)
        return feat * std + mean

# Cell 8: 训练/加载最大维度的 PCA 模型，自动维度搜索并绘图（动态适应不同下采样倍率）
import numpy as np

def accumulate_latent_stats_from_cache(node_name, pca_model, node_stats_dict, device):
    """从缓存特征计算隐空间的 sum_z, sum_zz, n_total"""
    mean_feat, std_feat = node_stats_dict[node_name]
    comps = pca_model["components"].to(device)
    mean_vec = pca_model["mean"].to(device)
    sum_z = None; sum_zz = None; n_total = 0

    node_cache_dir = os.path.join(cache_feat_dir, node_name)
    N = len(train_list)
    for idx in range(N):
        feat_down = torch.load(os.path.join(node_cache_dir, f"{idx}.pt"), map_location='cpu')
        norm_feat = normalize_feat(feat_down, mean_feat, std_feat)
        feat_flat = norm_feat.flatten(1).to(device)
        feat_c = feat_flat - mean_vec.unsqueeze(0)
        z = feat_c @ comps.T
        batch_sum = z.sum(dim=0).cpu()
        batch_zz = (z.T @ z).cpu()
        if sum_z is None:
            sum_z = batch_sum; sum_zz = batch_zz
        else:
            sum_z += batch_sum; sum_zz += batch_zz
        n_total += 1
        del feat_down, norm_feat, feat_flat, feat_c, z
    return sum_z, sum_zz, n_total

# Cell 9: 辅助函数定义（统一接口，支持下采样，返回各类 Dice）
import numpy as np
from scipy.optimize import minimize_scalar

class PCAEncoderDecoder:
    def __init__(self, components, mean, orig_shape):
        self.components = components
        self.mean = mean
        self.orig_shape = orig_shape   # 下采样后的形状 (C, H_down, W_down, D_down)
        self.latent_dim = components.shape[0]
    def encode(self, x):
        x_flat = x.flatten(1)
        z = (x_flat - self.mean.to(x.device)) @ self.components.T.to(x.device)
        return z
    def decode(self, z):
        x_flat = z @ self.components.to(z.device) + self.mean.to(z.device)
        return x_flat.reshape(-1, *self.orig_shape)
    def decode_residual(self, z):
        x_flat = z @ self.components.to(z.device)
        return x_flat.reshape(-1, *self.orig_shape)
    def to(self, device):
        self.components = self.components.to(device)
        self.mean = self.mean.to(device)
        return self

# 岭回归与 GCV（不变）
def _gcv_ridge(eigvals, V, Vt_C_mf, tss, N, d, lam):
    di = eigvals / (eigvals + lam)
    df_single = di.sum().item()
    df_total = d * df_single
    Vt_C_mf_row_sq = (Vt_C_mf ** 2).sum(dim=1)
    UTY2 = Vt_C_mf_row_sq / eigvals.clamp(min=1e-15)
    gain = (di ** 2) * UTY2
    rss = tss - gain.sum().item()
    rss = max(rss, 0.0)
    scale = 1.0 - df_total / (N * d)
    if scale <= 0: return 1e30
    gcv = (rss / (N * d)) / (scale ** 2)
    return gcv

def select_lambda_gcv(C_miss, C_mf, tss, N, d, device='cpu'):
    C_miss_np = C_miss.cpu().numpy()
    eigvals_np, V_np = np.linalg.eigh(C_miss_np)
    eigvals_np = eigvals_np[::-1].copy()
    V_np = V_np[:, ::-1].copy()
    eigvals = torch.from_numpy(eigvals_np).float().to(device)
    V = torch.from_numpy(V_np).float().to(device)
    Vt_C_mf = V.T @ C_mf.to(device)
    def objective(log_lam):
        lam = np.exp(log_lam)
        return _gcv_ridge(eigvals, V, Vt_C_mf, tss, N, d, lam)
    res = minimize_scalar(objective, bounds=(-13.8, 4.6), method='bounded')
    if res.success: return np.exp(res.x)
    else: return 0.01

def compute_correction_from_zs(z_miss, z_full, device='cpu'):
    N, d = z_miss.shape
    mu_miss = z_miss.mean(dim=0); mu_full = z_full.mean(dim=0)
    z_miss_c = z_miss - mu_miss
    delta_z_orig = z_full - z_miss
    mu_delta = mu_full - mu_miss
    delta_z_c = delta_z_orig - mu_delta
    C_miss = z_miss_c.T @ z_miss_c; C_mf = z_miss_c.T @ delta_z_c
    tss = delta_z_c.pow(2).sum().item()
    best_lam = select_lambda_gcv(C_miss, C_mf, tss, N, d, device=device)
    W = torch.linalg.solve(C_miss + best_lam * torch.eye(d, device=device), C_mf.to(device))
    b = mu_delta.to(device) - mu_miss.to(device) @ W
    delta_z_pred = z_miss.to(device) @ W + b.to(device)
    rss = (delta_z_orig.to(device) - delta_z_pred).pow(2).sum().item()
    mse = rss / (N * d)
    r2 = 1.0 - rss / tss if tss > 0 else 0.0
    reg_loss = best_lam * torch.sum(W ** 2).item()
    return W.cpu(), b.cpu().squeeze(0), best_lam, r2, mse, reg_loss

def dice_coefficient(pred_logits, target, num_classes=4, smooth=1e-5):
    pred = torch.softmax(pred_logits, dim=1).argmax(1)
    target = target.squeeze(1).long()
    dices = []
    for c in range(1, num_classes):
        pred_c = (pred == c)
        target_c = (target == c)
        inter = (pred_c & target_c).sum().float()
        dice = (2.*inter + smooth) / (pred_c.sum() + target_c.sum() + smooth)
        dices.append(dice)
    mean_dice = torch.stack(dices).mean()
    return mean_dice.item(), [d.item() for d in dices]

max_order = graph.num_levels - 1

for pattern_name, missing_indices in missing_patterns:
    logger.info(f"\n{'='*60}\nProcessing pattern: {pattern_name}\n{'='*60}")
    autoencoders_best = {}; corrections_best = {}; curve_data[pattern_name] = {}
    for order in range(max_order):
        if order not in order_to_nodes: continue
        nodes_in_order = order_to_nodes[order]
        logger.info(f"  Order {order}: nodes {nodes_in_order}")
        latent_data = collect_order_latent_vectors(
            train_loader_collect, missing_indices,
            order, order_to_nodes,
            autoencoders_max, autoencoders_best, corrections_best,
            node_stats, device)
        for node_name in nodes_in_order:
            # 动态计算该节点的 k_max
            D = int(np.prod(node_downsample_shapes[node_name]))
            k_max = min(L, D)
            if k_max < 1:
                continue
            save_path = os.path.join(corrections_dir, f"{pattern_name}_{node_name}.pt")
            if os.path.exists(save_path):
                logger.info(f"    {node_name}: correction file exists, loading.")
                corr_data = torch.load(save_path, map_location='cpu')
                delta_dice = corr_data.get("delta_dice", 0.0)
                corrections_best[node_name] = {"W": corr_data["W"], "b": corr_data["b"]}
                if delta_dice >= 0:
                    d = corr_data["W"].shape[0]
                    # 加载 PCA 模型时用对应的 k_max（注意文件名中的 k_max 是训练时的，这里我们重新计算 k_max 可能变小，但文件名是基于原始 fit 的 k_max，所以直接用 k_max 构造路径，因为文件就是用这个 k_max 保存的）
                    pca_path = os.path.join(SAVE_DIR, f"pca_latent{k_max}_{node_name}_{CONFIG_TAG}.pth")
                    model_max = torch.load(pca_path, map_location='cpu')
                    comp_d = model_max["components"][:d, :].clone()
                    ae = PCAEncoderDecoder(comp_d, model_max["mean"].clone(),
                                           node_downsample_shapes[node_name]).to(device)
                    autoencoders_best[node_name] = ae
                    logger.info(f"    {node_name}: activated (delta_dice={delta_dice:+.4f})")
                else: 
                    logger.info(f"    {node_name}: not activated (delta_dice={delta_dice:+.4f})")
                continue

            z_miss_max, z_full_max = latent_data[node_name]
            acc_without, _ = evaluate_corrected_mem(
                graph, val_loader_search, missing_indices,
                autoencoders_best, corrections_best,
                order_to_nodes, node_stats, {})
            logger.info(f"    {node_name} without correction: Dice={acc_without:.4f}")

            # 构建搜索候选
            candidates = list(range(SEARCH_START, k_max + 1, SEARCH_STEP))
            if candidates[-1] != k_max:
                candidates.append(k_max)
            acc_list = []; best_acc = -1.0; best_d = None; best_W = best_b = None
            for d in candidates:
                z_miss_d = z_miss_max[:, :d]; z_full_d = z_full_max[:, :d]
                W, b, lam, r2, mse, reg_loss = compute_correction_from_zs(z_miss_d, z_full_d, device='cpu')
                pca_path = os.path.join(SAVE_DIR, f"pca_latent{k_max}_{node_name}_{CONFIG_TAG}.pth")
                model_max = torch.load(pca_path, map_location='cpu')
                comp_d = model_max["components"][:d, :].clone()
                ae_d = PCAEncoderDecoder(comp_d, model_max["mean"].clone(),
                                         node_downsample_shapes[node_name]).to(device)
                temp_ae = {**autoencoders_best, node_name: ae_d}
                temp_corr = {**corrections_best, node_name: {"W": W, "b": b}}
                acc_mean, _ = evaluate_corrected_mem(
                    graph, val_loader_search, missing_indices,
                    temp_ae, temp_corr, order_to_nodes, node_stats, {node_name: ALPHA_EVAL})
                acc_list.append(acc_mean)
                logger.info(f"    {node_name} dim={d:4d}  Dice={acc_mean:.4f}  λ={lam:.4e}  R²={r2:.4f}  MSE={mse:.6f}  regloss={reg_loss:.6f}")
                if acc_mean > best_acc:
                    best_acc = acc_mean; best_d = d; best_W, best_b = W, b
                    torch.save({"W": best_W, "b": best_b, "delta_dice": 0.0}, save_path)
                    logger.info(f"    >>> New best dim {best_d} found, intermediate saved.")

            delta_dice = best_acc - acc_without
            torch.save({"W": best_W, "b": best_b, "delta_dice": delta_dice}, save_path)
            logger.info(f"    >> Final best dim {best_d} saved (delta_dice={delta_dice:+.4f}).")
            if delta_dice >= 0:
                pca_path = os.path.join(SAVE_DIR, f"pca_latent{k_max}_{node_name}_{CONFIG_TAG}.pth")
                best_comp = torch.load(pca_path, map_location='cpu')["components"][:best_d, :].clone()
                best_mean = torch.load(pca_path, map_location='cpu')["mean"].clone()
                best_ae = PCAEncoderDecoder(best_comp, best_mean,
                                            node_downsample_shapes[node_name]).to(device)
                autoencoders_best[node_name] = best_ae
                corrections_best[node_name] = {"W": best_W, "b": best_b}
                logger.info(f"    >> Activated {node_name} (dim={best_d})")
            else:
                logger.info(f"    >> NOT activated {node_name} (delta_dice <= 0)")
            curve_data[pattern_name][node_name] = (candidates, acc_list)
            del z_miss_max, z_full_max
            torch.cuda.empty_cache(); gc.collect()
    all_best_autoencoders[pattern_name] = autoencoders_best
    all_best_dims[pattern_name] = {n: ae.latent_dim for n, ae in autoencoders_best.items()}

torch.save(all_best_autoencoders, os.path.join(SAVE_DIR, f"all_best_autoencoders_{CONFIG_TAG}.pth"))
torch.save(all_best_dims, os.path.join(SAVE_DIR, f"all_best_dims_{CONFIG_TAG}.pth"))

    print(f"\n{'='*60}\nGreedy alpha search for {pattern_name}\n{'='*60}")
    active_nodes = [n for n in pattern_ae.keys() if n not in disabled_nodes]
    current_alpha_dict = {n:0.0 for n in active_nodes}
    best_alphas = {}
    pbar = tqdm(total=len(active_nodes)*len(alpha_candidates), desc=f'Greedy search {pattern_name}')
    for node_name in active_nodes:

        for a in alpha_candidates:
            test_dict = current_alpha_dict.copy(); test_dict[node_name] = a
            with open(os.devnull, 'w') as fnull, redirect_stderr(fnull):
                corr_mean, corr_class = evaluate_corrected(graph, val_loader_alpha, pattern_name, missing_indices,
                                                           pattern_ae, order_to_nodes, corrections_dir, node_stats,
                                                           alpha_dict=test_dict, disabled_nodes=disabled_nodes)
            if corr_mean > best_dice: best_dice = corr_mean; best_alpha = a
            pbar.update(1)
        current_alpha_dict[node_name] = best_alpha; best_alphas[node_name] = best_alpha
        tqdm.write(f"  {node_name}: best alpha = {best_alpha}, val Mean Dice = {best_dice:.4f}")
    pbar.close()
    final_mean, final_class = evaluate_corrected(graph, val_loader_alpha, pattern_name, missing_indices,
                                                 pattern_ae, order_to_nodes, corrections_dir, node_stats,
